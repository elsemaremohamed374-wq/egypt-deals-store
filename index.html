/* SMOOTH CATEGORY IMAGE SLIDER */ const slideImages = [ {src:"./electronics-banner.jpg",alt:"الإلكترونيات والتكنولوجيا"}, {src:"./fashion-banner.jpg",alt:"الملابس والأزياء"}, {src:"./appliances-banner.jpg",alt:"الأجهزة المنزلية"}, {src:"./makeup-banner.jpg",alt:"المكياج والتجميل"}, {src:"./accessories-banner.jpg",alt:"الإكسسوارات"} ]; const categorySlide = document.getElementById("categorySlide"); let slideIndex = 0; let sliderTimer = null; let isChanging = false; // تحميل الصور مسبقاً function preloadImage(src) { return new Promise((resolve) => { const img = new Image(); img.onload = () => resolve(src); img.onerror = () => resolve(null); img.src = src; }); } async function startCategorySlider() { if (!categorySlide) return; const loadedImages = await Promise.all( slideImages.map(async (item) => { const result = await preloadImage(item.src); return result ? item : null; }) ); const availableImages = loadedImages.filter(Boolean); if (availableImages.length < 2) return; // تحديد الصورة الحالية const currentSrc = categorySlide.getAttribute("src"); const currentIndex = availableImages.findIndex( item => item.src === currentSrc ); slideIndex = currentIndex >= 0 ? currentIndex : 0; // بدء التبديل بعد اكتمال التحميل sliderTimer = setInterval(() => { if (isChanging) return; isChanging = true; categorySlide.classList.add("fade"); setTimeout(() => { slideIndex = (slideIndex + 1) % availableImages.length; categorySlide.src = availableImages[slideIndex].src; categorySlide.alt = availableImages[slideIndex].alt; requestAnimationFrame(() => { requestAnimationFrame(() => { categorySlide.classList.remove("fade"); isChanging = false; }); }); }, 650); }, 3500); } startCategorySlider();
